#=======================================================
# Comandos de python
#=======================================================

# Crear un ambiente virtual
python3 -m venv .venv

# Activar ambiente virtual
source .venv/bin/activate

# Instalar librerías desde archivo "requirements"
pip3 install -r requirements.txt

#=======================================================
# Comandos de Git
#=======================================================
# Revisar status
git status

# Agregar archivo
git "nombre archivo" | ej: git add main.py

# Confirmar cambios
git add "nombre archivo" | ej de comentario: "backup"

# Cargar cambios al ecosistema
git push origin main
